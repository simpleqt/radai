## 一、概述

### 1.1 背景（规划动机 · 部分已缓解）

规划阶段假设系统仅依赖 API Key 明文传输；**现行**已具备 Redis Token、werkzeug 密码哈希与 API Key 表。仍待建设的能力包括：
- API Key 明文传输，易被中间人攻击截获
- 缺少请求签名验证，无法防止请求篡改
- 缺少 Token 过期和刷新机制
- 缺少细粒度的权限控制

### 1.2 目标

构建一套安全、可扩展的统一认证体系，支持：
- **API 调用**：服务间调用、第三方集成
- **MCP 调用**：Model Context Protocol 服务调用
- **网页端访问**：Web 门户、管理后台
- **移动端访问**：iOS/Android SDK、小程序

#### 2.1.2 请求签名算法

```python
import hmac
import hashlib
import time
import uuid
import json

def generate_signature(
    api_key: str,
    api_secret: str,
    method: str,
    path: str,
    body: dict | None = None,
    timestamp: int | None = None,
    nonce: str | None = None,
) -> dict:
    """
    生成 API 请求签名
    
    Args:
        api_key: API Key
        api_secret: API Secret
        method: HTTP 方法 (GET/POST/PUT/DELETE)
        path: 请求路径
        body: 请求体
        timestamp: 时间戳 (毫秒)
        nonce: 随机字符串
    
    Returns:
        包含签名的请求头
    """
    timestamp = timestamp or int(time.time() * 1000)
    nonce = nonce or str(uuid.uuid4())
    
    # 构建签名字符串
    body_str = json.dumps(body, separators=(',', ':')) if body else ""
    sign_str = f"{method}\n{path}\n{timestamp}\n{nonce}\n{body_str}"
    
    # HMAC-SHA256 签名
    signature = hmac.new(
        api_secret.encode('utf-8'),
        sign_str.encode('utf-8'),
        hashlib.sha256
    ).hexdigest()
    
    return {
        "X-API-Key": api_key,
        "X-Timestamp": str(timestamp),
        "X-Nonce": nonce,
        "X-Signature": signature,
        "Content-Type": "application/json"
    }
```

#### 2.1.3 服务端验证

```python
from fastapi import Request, HTTPException
from datetime import datetime
import time

class SignatureAuth:
    """签名认证中间件"""
    
    SIGNATURE_EXPIRE_SECONDS = 300  # 签名有效期 5 分钟
    NONCE_CACHE_TTL = 600  # Nonce 缓存 10 分钟
    
    async def verify(self, request: Request) -> dict:
        """验证请求签名"""
        api_key = request.headers.get("X-API-Key")
        timestamp = request.headers.get("X-Timestamp")
        nonce = request.headers.get("X-Nonce")
        signature = request.headers.get("X-Signature")
        
        # 1. 检查必要参数
        if not all([api_key, timestamp, nonce, signature]):
            raise HTTPException(401, "缺少认证参数")
        
        # 2. 检查时间戳 (防重放攻击)
        current_time = int(time.time() * 1000)
        request_time = int(timestamp)
        if abs(current_time - request_time) > self.SIGNATURE_EXPIRE_SECONDS * 1000:
            raise HTTPException(401, "请求已过期")
        
        # 3. 检查 Nonce (防重放攻击)
        if await self._is_nonce_used(nonce):
            raise HTTPException(401, "请求已被使用")
        
        # 4. 获取 API Secret
        api_secret = await self._get_api_secret(api_key)
        if not api_secret:
            raise HTTPException(401, "无效的 API Key")
        
        # 5. 验证签名
        body = await request.body()
        expected_signature = self._calculate_signature(
            api_key, api_secret, request.method, 
            request.url.path, timestamp, nonce, body
        )
        
        if not hmac.compare_digest(signature, expected_signature):
            raise HTTPException(401, "签名验证失败")
        
        # 6. 记录 Nonce 已使用
        await self._record_nonce(nonce)
        
        return {"api_key": api_key, "user_id": await self._get_user_id(api_key)}
```

#### 2.2.3 MCP Token 认证

```python
from pydantic import BaseModel
from typing import Optional
import jwt
from datetime import datetime, timedelta

class MCPTokenPayload(BaseModel):
    """MCP Token 载荷"""
    sub: str  # 服务标识
    aud: str  # 目标服务
    iat: int  # 签发时间
    exp: int  # 过期时间
    scope: list[str]  # 权限范围
    jti: str  # Token ID (防重放)

class MCPAuth:
    """MCP 认证管理"""
    
    def __init__(self, secret_key: str, algorithm: str = "HS256"):
        self.secret_key = secret_key
        self.algorithm = algorithm
    
    def generate_token(
        self,
        service_id: str,
        target_service: str,
        scopes: list[str],
        expires_hours: int = 24
    ) -> str:
        """生成 MCP Token"""
        now = datetime.utcnow()
        payload = {
            "sub": service_id,
            "aud": target_service,
            "iat": int(now.timestamp()),
            "exp": int((now + timedelta(hours=expires_hours)).timestamp()),
            "scope": scopes,
            "jti": str(uuid.uuid4())
        }
        return jwt.encode(payload, self.secret_key, algorithm=self.algorithm)
    
    async def verify_token(self, token: str, required_scope: str) -> MCPTokenPayload:
        """验证 MCP Token"""
        try:
            payload = jwt.decode(
                token, 
                self.secret_key, 
                algorithms=[self.algorithm],
                audience="mcp-server"
            )
            
            # 检查权限范围
            if required_scope not in payload.get("scope", []):
                raise HTTPException(403, "权限不足")
            
            # 检查 Token 是否已吊销
            if await self._is_token_revoked(payload["jti"]):
                raise HTTPException(401, "Token 已被吊销")
            
            return MCPTokenPayload(**payload)
            
        except jwt.ExpiredSignatureError:
            raise HTTPException(401, "Token 已过期")
        except jwt.InvalidTokenError as e:
            raise HTTPException(401, f"无效的 Token: {e}")
```

#### 2.3.2 JWT Token 结构

```python
from pydantic import BaseModel
from typing import Optional
import jwt
from datetime import datetime, timedelta
import secrets

class TokenPair(BaseModel):
    """Token 对"""
    access_token: str
    refresh_token: str
    token_type: str = "Bearer"
    expires_in: int  # 秒

class JWTAuth:
    """JWT 认证管理"""
    
    ACCESS_TOKEN_EXPIRE_MINUTES = 30  # 访问 Token 有效期
    REFRESH_TOKEN_EXPIRE_DAYS = 7     # 刷新 Token 有效期
    
    def __init__(
        self,
        secret_key: str,
        algorithm: str = "HS256",
        issuer: str = "agent-platform",
        audience: str = "web-client"
    ):
        self.secret_key = secret_key
        self.algorithm = algorithm
        self.issuer = issuer
        self.audience = audience
    
    def create_token_pair(
        self,
        user_id: str,
        username: str,
        roles: list[str],
        permissions: list[str]
    ) -> TokenPair:
        """创建 Token 对"""
        now = datetime.utcnow()
        
        # 访问 Token
        access_payload = {
            "sub": user_id,
            "username": username,
            "roles": roles,
            "permissions": permissions,
            "type": "access",
            "iat": int(now.timestamp()),
            "exp": int((now + timedelta(minutes=self.ACCESS_TOKEN_EXPIRE_MINUTES)).timestamp()),
            "iss": self.issuer,
            "aud": self.audience,
            "jti": secrets.token_urlsafe(16)
        }
        access_token = jwt.encode(access_payload, self.secret_key, algorithm=self.algorithm)
        
        # 刷新 Token
        refresh_payload = {
            "sub": user_id,
            "type": "refresh",
            "iat": int(now.timestamp()),
            "exp": int((now + timedelta(days=self.REFRESH_TOKEN_EXPIRE_DAYS)).timestamp()),
            "iss": self.issuer,
            "aud": self.audience,
            "jti": secrets.token_urlsafe(32)
        }
        refresh_token = jwt.encode(refresh_payload, self.secret_key, algorithm=self.algorithm)
        
        return TokenPair(
            access_token=access_token,
            refresh_token=refresh_token,
            expires_in=self.ACCESS_TOKEN_EXPIRE_MINUTES * 60
        )
    
    async def refresh_access_token(self, refresh_token: str) -> TokenPair:
        """刷新访问 Token"""
        try:
            payload = jwt.decode(
                refresh_token,
                self.secret_key,
                algorithms=[self.algorithm],
                audience=self.audience,
                issuer=self.issuer
            )
            
            if payload.get("type") != "refresh":
                raise HTTPException(401, "无效的刷新 Token")
            
            # 检查 Token 是否已吊销
            if await self._is_token_revoked(payload["jti"]):
                raise HTTPException(401, "Token 已被吊销")
            
            # 获取用户信息并生成新 Token 对
            user_id = payload["sub"]
            user_info = await self._get_user_info(user_id)
            
            # 将旧的刷新 Token 加入黑名单
            await self._revoke_token(payload["jti"])
            
            return self.create_token_pair(
                user_id=user_info["user_id"],
                username=user_info["username"],
                roles=user_info["roles"],
                permissions=user_info["permissions"]
            )
            
        except jwt.ExpiredSignatureError:
            raise HTTPException(401, "刷新 Token 已过期，请重新登录")
        except jwt.InvalidTokenError as e:
            raise HTTPException(401, f"无效的 Token: {e}")
```

```typescript
// 备选：使用 LocalStorage + 内存缓存 (防 CSRF)
class TokenManager {
  private accessToken: string | null = null;
  private refreshToken: string | null = null;
  
  setTokens(access: string, refresh: string): void {
    this.accessToken = access;
    this.refreshToken = refresh;
    localStorage.setItem('refresh_token', refresh);
  }
  
  getAccessToken(): string | null {
    return this.accessToken;
  }
  
  async refreshAccessToken(): Promise<string> {
    const response = await fetch('/api/auth/refresh', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.refreshToken}`
      }
    });
    
    const data = await response.json();
    this.accessToken = data.access_token;
    this.refreshToken = data.refresh_token;
    
    return this.accessToken;
  }

    this.accessToken = null;
    this.refreshToken = null;
    localStorage.removeItem('refresh_token');
  }
}
```

```python
    device_id: str
    device_name: str
    device_type: DeviceType
    os_version: str
    app_version: str
    push_token: Optional[str] = None  # FCM/APNs Token

class MobileTokenPayload(BaseModel):
    """移动端 Token 载荷"""
    sub: str  # 用户 ID
    device_id: str  # 设备 ID
    session_id: str  # 会话 ID
    iat: int
    exp: int
    type: str  # access/refresh

class MobileAuth:
    """移动端认证管理"""
    
    MAX_DEVICES_PER_USER = 5  # 每用户最大设备数
    SESSION_EXPIRE_DAYS = 30  # 会话有效期
    
    async def register_device(
        self,
        user_id: str,
        device_info: DeviceInfo
    ) -> str:
        """注册设备"""
        # 检查设备数量限制
        devices = await self._get_user_devices(user_id)
        if len(devices) >= self.MAX_DEVICES_PER_USER:
            # 移除最旧的设备
            await self._remove_oldest_device(user_id)
        
        # 注册设备
        device_token = secrets.token_urlsafe(32)
        await self._save_device(user_id, device_info, device_token)
        
        return device_token
    
    async def login(
        self,
        username: str,
        password: str,
        device_info: DeviceInfo,
        mfa_code: Optional[str] = None
    ) -> TokenPair:
        """移动端登录"""
        # 1. 验证用户凭证
        user = await self._verify_credentials(username, password)
        if not user:
            raise HTTPException(401, "用户名或密码错误")
        
        # 2. 检查是否需要 MFA
        if user.mfa_enabled:
            if not mfa_code:
                raise HTTPException(200, "需要多因素认证", {"mfa_required": True})
            if not await self._verify_mfa(user.user_id, mfa_code):
                raise HTTPException(401, "验证码错误")
        
        # 3. 注册/更新设备
        await self.register_device(user.user_id, device_info)
        
        # 4. 创建会话
        session_id = str(uuid.uuid4())
        await self._create_session(user.user_id, session_id, device_info)
        
        # 5. 生成 Token
        return self._create_mobile_token_pair(
            user_id=user.user_id,
            device_id=device_info.device_id,
            session_id=session_id,
            roles=user.roles,
            permissions=user.permissions
        )
    
    def _create_mobile_token_pair(
        self,
        user_id: str,
        device_id: str,
        session_id: str,
        roles: list[str],
        permissions: list[str]
    ) -> TokenPair:
        """创建移动端 Token 对"""
        now = datetime.utcnow()
        
        # 访问 Token (较短有效期)
        access_payload = {
            "sub": user_id,
            "device_id": device_id,
            "session_id": session_id,
            "roles": roles,
            "permissions": permissions,
            "type": "access",
            "iat": int(now.timestamp()),
            "exp": int((now + timedelta(hours=2)).timestamp()),
            "iss": self.issuer,
            "aud": "mobile-client"
        }
        
        # 刷新 Token (较长有效期)
        refresh_payload = {
            "sub": user_id,
            "device_id": device_id,
            "session_id": session_id,
            "type": "refresh",
            "iat": int(now.timestamp()),
            "exp": int((now + timedelta(days=self.SESSION_EXPIRE_DAYS)).timestamp()),
            "iss": self.issuer,
            "aud": "mobile-client"
        }
        
        return TokenPair(
            access_token=jwt.encode(access_payload, self.secret_key, self.algorithm),
            refresh_token=jwt.encode(refresh_payload, self.secret_key, self.algorithm),
            expires_in=2 * 60 * 60  # 2 小时
        )
```

### 3.1 认证中间件实现

```python
from fastapi import FastAPI, Request, HTTPException, Depends
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from starlette.middleware.base import BaseHTTPMiddleware
from typing import Optional, Callable
from enum import Enum
import re

class AuthType(str, Enum):
    """认证类型"""
    JWT = "jwt"           # 网页端/移动端
    SIGNATURE = "signature"  # API 调用
    MCP_TOKEN = "mcp"     # MCP 调用
    API_KEY = "api_key"   # 简单 API Key (向后兼容)

class AuthContext(BaseModel):
    """认证上下文"""
    auth_type: AuthType
    user_id: str
    username: Optional[str] = None
    roles: list[str] = []
    permissions: list[str] = []
    device_id: Optional[str] = None
    session_id: Optional[str] = None
    api_key: Optional[str] = None
    rate_limit: int = 100

class UnifiedAuthMiddleware(BaseHTTPMiddleware):
    """统一认证中间件"""
    
    # 不需要认证的路径
    PUBLIC_PATHS = [
        "/health",
        "/metrics",
        "/docs",
        "/openapi.json",
        "/auth/login",
        "/auth/register",
        "/auth/refresh",
        "/auth/forgot-password",
    ]
    
    # 路径与认证类型映射
    AUTH_TYPE_MAP = {
        "/api/v1/": AuthType.SIGNATURE,      # API 调用
        "/mcp/": AuthType.MCP_TOKEN,         # MCP 调用
        "/web/": AuthType.JWT,               # 网页端
        "/mobile/": AuthType.JWT,            # 移动端
    }
    
    def __init__(self, app: FastAPI):
        super().__init__(app)
        self.jwt_auth = JWTAuth(...)
        self.signature_auth = SignatureAuth()
        self.mcp_auth = MCPAuth(...)
    
    async def dispatch(self, request: Request, call_next: Callable):
        """处理请求"""
        path = request.url.path
        
        # 检查是否为公开路径
        if self._is_public_path(path):
            return await call_next(request)
        
        # 确定认证类型
        auth_type = self._determine_auth_type(path)
        
        # 执行认证
        try:
            auth_context = await self._authenticate(request, auth_type)
            request.state.auth = auth_context
        except HTTPException as e:
            return JSONResponse(
                status_code=e.status_code,
                content={"error": e.detail}
            )
        
        # 检查权限
        if not await self._check_permission(auth_context, path, request.method):
            return JSONResponse(
                status_code=403,
                content={"error": "权限不足"}
            )
        
        # 记录审计日志
        await self._log_access(auth_context, request)
        
        return await call_next(request)
    
    def _is_public_path(self, path: str) -> bool:
        """检查是否为公开路径"""
        for public_path in self.PUBLIC_PATHS:
            if path.startswith(public_path):
                return True
        return False
    
    def _determine_auth_type(self, path: str) -> AuthType:
        """确定认证类型"""
        for prefix, auth_type in self.AUTH_TYPE_MAP.items():
            if path.startswith(prefix):
                return auth_type
        return AuthType.JWT  # 默认 JWT
    
    async def _authenticate(
        self, 
        request: Request, 
        auth_type: AuthType
    ) -> AuthContext:
        """执行认证"""
        if auth_type == AuthType.JWT:
            return await self._jwt_authenticate(request)
        elif auth_type == AuthType.SIGNATURE:
            return await self._signature_authenticate(request)
        elif auth_type == AuthType.MCP_TOKEN:
            return await self._mcp_authenticate(request)
        elif auth_type == AuthType.API_KEY:
            return await self._api_key_authenticate(request)
        else:
            raise HTTPException(401, "未知的认证类型")
    
    async def _jwt_authenticate(self, request: Request) -> AuthContext:
        """JWT 认证"""
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith("Bearer "):
            raise HTTPException(401, "缺少认证 Token")
        
        token = auth_header[7:]
        payload = await self.jwt_auth.verify_token(token)
        
        return AuthContext(
            auth_type=AuthType.JWT,
            user_id=payload["sub"],
            username=payload.get("username"),
            roles=payload.get("roles", []),
            permissions=payload.get("permissions", []),
            device_id=payload.get("device_id"),
            session_id=payload.get("session_id")
        )
    
    async def _signature_authenticate(self, request: Request) -> AuthContext:
        """签名认证"""
        result = await self.signature_auth.verify(request)
        
        return AuthContext(
            auth_type=AuthType.SIGNATURE,
            user_id=result["user_id"],
            api_key=result["api_key"],
            rate_limit=result.get("rate_limit", 100)
        )
    
    async def _mcp_authenticate(self, request: Request) -> AuthContext:
        """MCP Token 认证"""
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith("Bearer "):
            raise HTTPException(401, "缺少 MCP Token")
        
        token = auth_header[7:]
        payload = await self.mcp_auth.verify_token(token, "mcp:call")
        
        return AuthContext(
            auth_type=AuthType.MCP_TOKEN,
            user_id=payload.sub,
            roles=["mcp_service"],
            permissions=["mcp:call", "mcp:tools", "mcp:resources"]
        )
```

```python
from functools import wraps
from typing import Callable

def require_permission(permission: str):
    """权限装饰器"""
    def decorator(func: Callable):
        @wraps(func)
        async def wrapper(*args, auth: AuthContext = None, **kwargs):
            if not auth:
                raise HTTPException(401, "未认证")
            
            if permission not in auth.permissions:
                raise HTTPException(403, f"缺少权限: {permission}")
            
            return await func(*args, auth=auth, **kwargs)
        return wrapper
    return decorator

def require_role(role: str):
    """角色装饰器"""

        @wraps(func)
        async def wrapper(*args, auth: AuthContext = None, **kwargs):
            if not auth:
                raise HTTPException(401, "未认证")
            
            if role not in auth.roles:
                raise HTTPException(403, f"需要角色: {role}")
            
            return await func(*args, auth=auth, **kwargs)
        return wrapper
    return decorator
```
